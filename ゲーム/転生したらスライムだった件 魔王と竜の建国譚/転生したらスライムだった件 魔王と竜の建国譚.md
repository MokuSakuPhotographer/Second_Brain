> [!CAUTION] 注意
> 覚醒し、星のレベルが上がっているものはその星の数にしてある。そのキャラの凸数は覚醒後、何凸しているかを表す。
> 育成済みにチェックが入っているキャラはレベルが最大ということを表す。スキルや才能解放までMAXにしてあるわけではない。
> 凸数が空欄のキャラは0凸である。

const body = table.createEl("tbody");

async function save(name, update) {
    const file = app.vault.getAbstractFileByPath(path);
    await app.vault.process(file, content => {
        const lines = content.split(/(?<=\n)/);
        const start = lines.findIndex(line => line.trim() === "### " + kind);
        const end = lines.findIndex((line, index) => index > start && line.trim() === "```");
        const matches = lines.flatMap((line, index) =>
            index > start && index < end && line.trimEnd().match(entryPattern)?.[4] === name ? [index] : []
        );
        if (start < 0 || matches.length !== 1) throw new Error("キャラの行を特定できません");
        const index = matches[0];
        lines[index] = update(lines[index]);
        return lines.join("");
    });
}

async function saveControl(control, rollback, update) {
    control.disabled = true;
    try {
        await update();
    } catch (error) {
        rollback();
        new Notice("キャラ情報を保存できませんでした");
        console.error(error);
    } finally {
        control.disabled = false;
    }
}

function renderRow(entry) {
    const [, status, initialRarity = "", initialAscension = "", name] = entry.trimEnd().match(entryPattern);
    const row = body.createEl("tr");
    row.createEl("td", { text: name });

    const rarity = row.createEl("td").createEl("select");
    rarity.setAttribute("aria-label", name + "のレアリティ");
    options(rarity, [["", "未設定"], ["7", "☆7"], ["6", "☆6"], ["5", "☆5"], ["4", "☆4"]]);
    rarity.value = initialRarity;
    let savedRarity = initialRarity;
    rarity.addEventListener("change", () => {
        const value = rarity.value;
        return saveControl(rarity, () => { rarity.value = savedRarity; }, async () => {
            await save(name, line => line.replace(/^(\[[ x]\] )(?:☆[4567] )?/, (_, prefix) =>
                prefix + (value ? "☆" + value + " " : "")
            ));
            savedRarity = value;
            sortRows();
        });
    });

    const ascension = row.createEl("td").createEl("input");
    ascension.type = "number";
    ascension.min = "0";
    ascension.step = "1";
    ascension.style.width = "5em";
    ascension.setAttribute("aria-label", name + "の凸数");
    ascension.value = initialAscension;
    let savedAscension = initialAscension;
    ascension.addEventListener("change", () => {
        const value = ascension.value;
        if (!ascension.validity.valid || (value && (!/^\d+$/.test(value) || !Number.isSafeInteger(Number(value))))) {
            ascension.value = savedAscension;
            return;
        }
        return saveControl(ascension, () => { ascension.value = savedAscension; }, async () => {
            await save(name, line => line.replace(/^(\[[ x]\] (?:☆[4567] )?)(?:凸\d+ )?/, (_, prefix) =>
                prefix + (value === "" ? "" : "凸" + Number(value) + " ")
            ));
            savedAscension = ascension.value = value === "" ? "" : String(Number(value));
        });
    });

    const checkbox = row.createEl("td").createEl("input", { type: "checkbox" });
    checkbox.type = "checkbox";
    checkbox.checked = status === "x";
    checkbox.setAttribute("aria-label", name + "の育成済み");
    checkbox.addEventListener("change", () => {
        const checked = checkbox.checked;
        return saveControl(checkbox, () => { checkbox.checked = !checked; },
            () => save(name, line => "[" + (checked ? "x" : " ") + line.slice(2)));
    });
    return { row, rarity };
}

const rows = entries.map(renderRow);

function sortRows() {
    const order = sort.value;
    const sorted = order === "original" ? rows : rows.slice().sort((a, b) => {
        const left = a.rarity.value;
        const right = b.rarity.value;
        if (!left || !right) return Number(!left) - Number(!right);
        return order === "descending" ? Number(right) - Number(left) : Number(left) - Number(right);
    });
    body.append(...sorted.map(({ row }) => row));
}
sort.addEventListener("change", () => {
    localStorage.setItem(sortKey, sort.value);
    sortRows();
});
sortRows();
```

        return saveControl(rarity, () => { rarity.value = savedRarity; }, async () => {
            await save(name, line => line.replace(/^(\[[ x]\] )(?:☆[4567] )?/, (_, prefix) =>
                prefix + (value ? "☆" + value + " " : "")
            ));
            savedRarity = value;
            sortRows();
        });
    });

    const ascension = row.createEl("td").createEl("input");
    ascension.type = "number";
    ascension.min = "0";
    ascension.step = "1";
    ascension.style.width = "5em";
    ascension.setAttribute("aria-label", name + "の凸数");
    ascension.value = initialAscension;
    let savedAscension = initialAscension;
    ascension.addEventListener("change", () => {
        const value = ascension.value;
        if (!ascension.validity.valid || (value && (!/^\d+$/.test(value) || !Number.isSafeInteger(Number(value))))) {
            ascension.value = savedAscension;
            return;
        }
        return saveControl(ascension, () => { ascension.value = savedAscension; }, async () => {
            await save(name, line => line.replace(/^(\[[ x]\] (?:☆[4567] )?)(?:凸\d+ )?/, (_, prefix) =>
                prefix + (value === "" ? "" : "凸" + Number(value) + " ")
            ));
            savedAscension = ascension.value = value === "" ? "" : String(Number(value));
        });
    });

    const checkbox = row.createEl("td").createEl("input", { type: "checkbox" });
    checkbox.type = "checkbox";
    checkbox.checked = status === "x";
    checkbox.setAttribute("aria-label", name + "の育成済み");
    checkbox.addEventListener("change", () => {
        const checked = checkbox.checked;
        return saveControl(checkbox, () => { checkbox.checked = !checked; },
            () => save(name, line => "[" + (checked ? "x" : " ") + line.slice(2)));
    });
    return { row, rarity };
}

const rows = entries.map(renderRow);

function sortRows() {
    const order = sort.value;
    const sorted = order === "original" ? rows : rows.slice().sort((a, b) => {
        const left = a.rarity.value;
        const right = b.rarity.value;
        if (!left || !right) return Number(!left) - Number(!right);
        return order === "descending" ? Number(right) - Number(left) : Number(left) - Number(right);
    });
    body.append(...sorted.map(({ row }) => row));
}
sort.addEventListener("change", () => {
    localStorage.setItem(sortKey, sort.value);
    sortRows();
});
